   145→export default function ClientDetail({ clientId }: Props) {
   146→  const { user, view, setView, selectClient } = useStore()
   147→  const isAdmin = user?.role === 'admin'
   148→  const [client, setClient] = useState<ClientData | null>(null)
   149→  const [loading, setLoading] = useState(true)
   150→  const [activeTab, setActiveTab] = useState(view === 'sessions' ? 'sessions' : view === 'questionnaire' ? 'questionnaire' : view === 'training' ? 'training' : view === 'progress' ? 'progress' : 'profile')
   151→  const [editMode, setEditMode] = useState(false)
   152→  const [editForm, setEditForm] = useState<Record<string, string>>({})
   153→  const [trainerNote, setTrainerNote] = useState('')
   154→
   155→  // Training state
   156→  const [showTrainingForm, setShowTrainingForm] = useState(false)
   157→  const [trainingForm, setTrainingForm] = useState<Training>({ id: '', name: '', isActive: true, days: [] })
   158→  const [showSessionForm, setShowSessionForm] = useState(false)
   159→  const [sessionForm, setSessionForm] = useState<Session>({ id: '', date: new Date().toISOString(), exercises: [] })
   160→  const [showQuestionnaire, setShowQuestionnaire] = useState(false)
   161→  const [qForm, setQForm] = useState<Record<string, number>>({})
   162→  const [qNotes, setQNotes] = useState('')
   163→  const [showMeasurement, setShowMeasurement] = useState(false)
   164→  const [measForm, setMeasForm] = useState<Record<string, string>>({})
   165→
   166→  useEffect(() => {
   167→    let cancelled = false
   168→    apiGet(`/api/clients/${clientId}`).then((data) => {
   169→      if (cancelled) return
   170→      setClient(data)
   171→      if (data) {
   172→        setEditForm({
   173→          name: data.name || '',
   174→          age: data.age?.toString() || '',
   175→          height: data.height?.toString() || '',
   176→          weight: data.weight?.toString() || '',
   177→          goal: data.goal || '',
   178→          experienceLevel: data.experienceLevel || '',
   179→          injuries: data.injuries || '',
   180→          weeklyAvailability: data.weeklyAvailability || '',
   181→          notes: data.notes || '',
   182→        })
   183→      }
   184→    }).catch(() => {}).finally(() => {
   185→      if (!cancelled) setLoading(false)
   186→    })
   187→    return () => { cancelled = true }
   188→  }, [clientId])
   189→
   190→  const reloadClient = () => {
   191→    apiGet(`/api/clients/${clientId}`).then((data) => {
   192→      setClient(data)
   193→      if (data) {
   194→        setEditForm({
   195→          name: data.name || '',
   196→          age: data.age?.toString() || '',
   197→          height: data.height?.toString() || '',
   198→          weight: data.weight?.toString() || '',
   199→          goal: data.goal || '',
   200→          experienceLevel: data.experienceLevel || '',
   201→          injuries: data.injuries || '',
   202→          weeklyAvailability: data.weeklyAvailability || '',
   203→          notes: data.notes || '',
   204→        })
   205→      }
   206→    }).catch(() => {})
   207→  }
   208→
   209→  const saveProfile = async () => {
   210→    try {
   211→      await apiPut(`/api/clients/${clientId}`, {
   212→        name: editForm.name,
   213→        age: editForm.age ? parseInt(editForm.age) : null,
   214→        height: editForm.height ? parseFloat(editForm.height) : null,
   215→        weight: editForm.weight ? parseFloat(editForm.weight) : null,
   216→        goal: editForm.goal || null,
   217→        experienceLevel: editForm.experienceLevel || null,
   218→        injuries: editForm.injuries || null,
   219→        weeklyAvailability: editForm.weeklyAvailability || null,
   220→        notes: editForm.notes || null,
   221→      })
   222→      setEditMode(false)
   223→      reloadClient()
   224→    } catch {}
   225→  }
   226→
   227→  const addTrainerNote = async () => {
   228→    if (!trainerNote.trim()) return
   229→    try {
   230→      await apiPost(`/api/clients/${clientId}`, { action: 'addNote', content: trainerNote })
   231→      setTrainerNote('')
   232→      reloadClient()
   233→    } catch {}
   234→  }
   235→
   236→  // Training helpers
   237→  const addTrainingDay = () => {
   238→    const num = trainingForm.days.length + 1
   239→    setTrainingForm({
   240→      ...trainingForm,
   241→      days: [...trainingForm.days, { dayNumber: num, dayName: `Día ${num}`, exercises: [] }],
   242→    })
   243→  }
   244→
   245→  const addExerciseToDay = (dayIdx: number) => {
   246→    const days = [...trainingForm.days]
   247→    days[dayIdx] = {
   248→      ...days[dayIdx],
   249→      exercises: [...days[dayIdx].exercises, { exerciseName: '', order: days[dayIdx].exercises.length, prescribedSets: 3, prescribedReps: '8-12', prescribedRir: '1-2' }],
   250→    }
   251→    setTrainingForm({ ...trainingForm, days })
   252→  }
   253→
   254→  const updateExercise = (dayIdx: number, exIdx: number, field: string, value: string | number) => {
   255→    const days = [...trainingForm.days]
   256→    const exercises = [...days[dayIdx].exercises]
   257→    exercises[exIdx] = { ...exercises[exIdx], [field]: value }
   258→    days[dayIdx] = { ...days[dayIdx], exercises }
   259→    setTrainingForm({ ...trainingForm, days })
   260→  }
   261→
   262→  const removeExercise = (dayIdx: number, exIdx: number) => {
   263→    const days = [...trainingForm.days]
   264→    days[dayIdx] = { ...days[dayIdx], exercises: days[dayIdx].exercises.filter((_, i) => i !== exIdx) }
   265→    setTrainingForm({ ...trainingForm, days })
   266→  }
   267→
   268→  const removeTrainingDay = (dayIdx: number) => {
   269→    const days = trainingForm.days.filter((_, i) => i !== dayIdx).map((d, i) => ({ ...d, dayNumber: i + 1 }))
   270→    setTrainingForm({ ...trainingForm, days })
   271→  }
   272→
   273→  const saveTraining = async () => {
   274→    try {
   275→      const body = {
   276→        name: trainingForm.name,
   277→        mesocycle: trainingForm.mesocycle || 1,
   278→        week: trainingForm.week || 1,
   279→        clientId,
   280→        days: trainingForm.days.map((d) => ({
   281→          dayNumber: d.dayNumber,
   282→          dayName: d.dayName,
   283→          exercises: d.exercises.map((e) => ({
   284→            exerciseName: e.exerciseName,
   285→            order: e.order,
   286→            prescribedSets: e.prescribedSets,
   287→            prescribedReps: e.prescribedReps,
   288→            prescribedRir: e.prescribedRir,
   289→            prescribedRpe: e.prescribedRpe,
   290→            prescribedRest: e.prescribedRest,
   291→            prescribedTempo: e.prescribedTempo,
   292→            prescribedWeight: e.prescribedWeight,
   293→            notes: e.notes,
   294→          })),
   295→        })),
   296→      }
   297→      if (trainingForm.id) {
   298→        await apiPut(`/api/trainings/${trainingForm.id}`, body)
   299→      } else {
   300→        await apiPost('/api/trainings', body)
   301→      }
   302→      setShowTrainingForm(false)
   303→      setTrainingForm({ id: '', name: '', isActive: true, days: [] })
   304→      reloadClient()
   305→    } catch {}
   306→  }
   307→
   308→  const duplicateTraining = async (t: Training) => {
   309→    try {
   310→      await apiGet(`/api/trainings/${t.id}?duplicate=true`)
   311→      reloadClient()
   312→    } catch {}
   313→  }
   314→
   315→  const deleteTraining = async (id: string) => {
   316→    try {
   317→      await apiDelete(`/api/trainings/${id}`)
   318→      reloadClient()
   319→    } catch {}
   320→  }
   321→
   322→  // Session helpers
   323→  const startSession = (dayIdx: number) => {
   324→    const day = client?.trainings.find(t => t.isActive)?.days[dayIdx]
   325→    if (!day) return
   326→    setSessionForm({
   327→      id: '', date: new Date().toISOString(),
   328→      exercises: day.exercises.map((e) => ({
   329→        exerciseName: e.exerciseName,
   330→        order: e.order,
   331→        sets: Array.from({ length: e.prescribedSets || 3 }, (_, i) => ({ setNumber: i + 1 })),
   332→      })),
   333→    })
   334→    setShowSessionForm(true)
   335→  }
   336→
   337→  const addSetToExercise = (exIdx: number) => {
   338→    const exercises = [...sessionForm.exercises]
   339→    const sets = [...exercises[exIdx].sets]
   340→    sets.push({ setNumber: sets.length + 1 })
   341→    exercises[exIdx] = { ...exercises[exIdx], sets }
   342→    setSessionForm({ ...sessionForm, exercises })
   343→  }
   344→
   345→  const updateSessionSet = (exIdx: number, setIdx: number, field: string, value: number | undefined) => {
   346→    const exercises = [...sessionForm.exercises]
   347→    const sets = [...exercises[exIdx].sets]
   348→    sets[setIdx] = { ...sets[setIdx], [field]: value }
   349→    exercises[exIdx] = { ...exercises[exIdx], sets }
   350→    setSessionForm({ ...sessionForm, exercises })
   351→  }
   352→
   353→  const saveSession = async () => {
   354→    try {
   355→      const activeTraining = client?.trainings.find(t => t.isActive)
   356→      await apiPost('/api/sessions', {
   357→        clientId,
   358→        trainingId: activeTraining?.id,
   359→        notes: sessionForm.notes,
   360→        rpe: sessionForm.rpe,
   361→        exercises: sessionForm.exercises.map((e) => ({
   362→          exerciseName: e.exerciseName,
   363→          order: e.order,
   364→          sets: e.sets.map((s) => ({
   365→            setNumber: s.setNumber,
   366→            reps: s.reps,
   367→            weight: s.weight,
   368→            rir: s.rir,
   369→            rpe: s.rpe,
   370→          })),
   371→        })),
   372→      })
   373→      setShowSessionForm(false)
   374→      setSessionForm({ id: '', date: new Date().toISOString(), exercises: [] })
   375→      reloadClient()
   376→    } catch {}
   377→  }
   378→
   379→  // Questionnaire
   380→  const saveQuestionnaire = async () => {
   381→    try {
   382→      await apiPost('/api/questionnaires', {
   383→        clientId,
   384→        sleepQuality: qForm.sleepQuality,
   385→        fatigue: qForm.fatigue,
   386→        stress: qForm.stress,
   387→        motivation: qForm.motivation,
   388→        pain: qForm.pain,
   389→        energy: qForm.energy,
   390→        hunger: qForm.hunger,
   391→        adherence: qForm.adherence,
   392→        trainingFeeling: qForm.trainingFeeling,
   393→        notes: qNotes,
   394→      })
   395→      setShowQuestionnaire(false)
   396→      setQForm({})
   397→      setQNotes('')
   398→      reloadClient()
   399→    } catch {}
   400→  }
   401→
   402→  // Measurement
   403→  const saveMeasurement = async () => {
   404→    try {
   405→      await apiPost('/api/measurements', {
   406→        clientId,
   407→        weight: measForm.weight ? parseFloat(measForm.weight) : null,
   408→        chest: measForm.chest ? parseFloat(measForm.chest) : null,
   409→        waist: measForm.waist ? parseFloat(measForm.waist) : null,
   410→        hips: measForm.hips ? parseFloat(measForm.hips) : null,
   411→      })
   412→      setShowMeasurement(false)
   413→      setMeasForm({})
   414→      reloadClient()
   415→    } catch {}
   416→  }
   417→
   418→  const deleteClient = async () => {
   419→    if (!confirm('¿Eliminar este cliente? Esta acción no se puede deshacer.')) return
   420→    try {
   421→      await apiDelete(`/api/clients/${clientId}`)
   422→      selectClient(null)
   423→      setView('clients')
   424→    } catch {}
   425→  }
   426→
   427→  const statusConfig = {
   428→    green: { label: 'Correcto', color: '#22c55e', bg: 'bg-green-50' },
   429→    yellow: { label: 'Revisar', color: '#eab308', bg: 'bg-yellow-50' },
   430→    red: { label: 'Atención', color: '#ef4444', bg: 'bg-red-50' },
   431→  }
   432→
   433→  const getScoreColor = (score?: number) => {
   434→    if (!score) return 'text-muted-foreground'
   435→    if (score >= 70) return 'text-green-600'
   436→    if (score >= 40) return 'text-yellow-600'
   437→    return 'text-red-600'
   438→  }
   439→
   440→  const getScoreBg = (score?: number) => {
   441→    if (!score) return 'bg-muted'
   442→    if (score >= 70) return 'bg-green-500'
   443→    if (score >= 40) return 'bg-yellow-500'
   444→    return 'bg-red-500'
   445→  }
   446→
   447→  if (loading) return <div className="flex items-center justify-center h-64 text-muted-foreground">Cargando...</div>
   448→  if (!client) return <div className="text-center py-12 text-muted-foreground">Cliente no encontrado</div>
   449→
   450→  const sc = statusConfig[client.status as keyof typeof statusConfig] || statusConfig.red
   451→  const latestQ = client.questionnaires?.[0]
   452→  const weightData = client.measurements?.filter(m => m.weight).map(m => ({ date: new Date(m.date).toLocaleDateString('es-ES', { day: '2-digit', month: 'short' }), peso: m.weight })) || []
   453→  const volumeData = client.sessions?.slice(-10).map(s => ({ date: new Date(s.date).toLocaleDateString('es-ES', { day: '2-digit', month: 'short' }), volumen: s.totalVolume || 0 })) || []
   454→
   455→  return (
   456→    <div className="space-y-6">
   457→      {/* Header */}
   458→      <div className="flex items-center gap-4 flex-wrap">
   459→        <Button variant="ghost" size="icon" onClick={() => { selectClient(null); setView('clients') }}>
   460→          <ArrowLeft className="h-5 w-5" />
   461→        </Button>
   462→        <div className="flex-1 min-w-0">
   463→          <div className="flex items-center gap-3 flex-wrap">
   464→            <h1 className="text-2xl font-light tracking-wider text-[#2D4A3E]">{client.name}</h1>
   465→            <Badge variant="outline" className="text-xs" style={{ borderColor: sc.color, color: sc.color }}>{sc.label}</Badge>
   466→          </div>
   467→          <p className="text-sm text-muted-foreground mt-1">{client.goal || 'Sin objetivo'}{client.weight ? ` · ${client.weight} kg` : ''}</p>
   468→        </div>
   469→        {isAdmin && (
   470→          <Button variant="outline" size="sm" className="text-red-500 border-red-200 hover:bg-red-50" onClick={deleteClient}>
   471→            <Trash2 className="h-4 w-4 mr-1" /> Eliminar
   472→          </Button>
   473→        )}
   474→      </div>
   475→
   476→      {/* Tabs */}
   477→      <div className="bg-white shadow-sm p-1 h-auto flex-wrap gap-1 flex rounded-lg">
   478→        {[
   479→          { value: 'profile', label: 'Perfil', icon: <Activity className="h-3.5 w-3.5 mr-1.5" /> },
   480→          { value: 'training', label: 'Entrenamiento', icon: <FileText className="h-3.5 w-3.5 mr-1.5" /> },
   481→          { value: 'sessions', label: 'Sesiones', icon: <ClipboardList className="h-3.5 w-3.5 mr-1.5" /> },
   482→          { value: 'questionnaire', label: 'Cuestionario', icon: <BarChart3 className="h-3.5 w-3.5 mr-1.5" /> },
   483→          { value: 'progress', label: 'Progreso', icon: <BarChart3 className="h-3.5 w-3.5 mr-1.5" /> },
   484→        ].map((tab) => (
   485→          <button
   486→            key={tab.value}
   487→            onClick={() => setActiveTab(tab.value)}
   488→            className={`inline-flex items-center gap-1.5 rounded-md px-3 py-2 text-xs font-medium transition-colors ${activeTab === tab.value ? 'bg-[#2D4A3E] text-white shadow-sm' : 'text-muted-foreground hover:text-foreground hover:bg-muted'}`}
   489→          >
   490→            {tab.icon}
   491→            {tab.label}
   492→          </button>
   493→        ))}
   494→      </div>
   495→
   496→        {/* PROFILE TAB */}
   497→      {activeTab === 'profile' && <div className="space-y-6 mt-6">
   498→          <div className="grid md:grid-cols-2 gap-6">
   499→            <Card className="border-0 shadow-sm">
   500→              <CardHeader className="pb-3">
   501→                <div className="flex items-center justify-between">
   502→                  <CardTitle className="text-xs font-medium text-muted-foreground uppercase tracking-wider">Datos personales</CardTitle>
   503→                  {isAdmin && (
   504→                    <Button variant="ghost" size="sm" onClick={() => setEditMode(!editMode)}>
   505→                      <Edit3 className="h-3.5 w-3.5 mr-1" /> {editMode ? 'Cancelar' : 'Editar'}
   506→                    </Button>
   507→                  )}
   508→                </div>
   509→              </CardHeader>
   510→              <CardContent>
   511→                {editMode && isAdmin ? (
   512→                  <div className="space-y-3">
   513→                    <div><Label className="text-xs uppercase">Nombre</Label><Input value={editForm.name} onChange={(e) => setEditForm(f => ({...f, name: e.target.value}))} className="mt-1" /></div>
   514→                    <div className="grid grid-cols-2 gap-3">
   515→                      <div><Label className="text-xs uppercase">Edad</Label><Input type="number" value={editForm.age} onChange={(e) => setEditForm(f => ({...f, age: e.target.value}))} className="mt-1" /></div>
   516→                      <div><Label className="text-xs uppercase">Altura (cm)</Label><Input type="number" value={editForm.height} onChange={(e) => setEditForm(f => ({...f, height: e.target.value}))} className="mt-1" /></div>
   517→                      <div><Label className="text-xs uppercase">Peso (kg)</Label><Input type="number" step="0.1" value={editForm.weight} onChange={(e) => setEditForm(f => ({...f, weight: e.target.value}))} className="mt-1" /></div>
   518→                      <div><Label className="text-xs uppercase">Objetivo</Label>
   519→                        <Select value={editForm.goal} onValueChange={(v) => setEditForm(f => ({...f, goal: v}))}>
   520→                          <SelectTrigger className="mt-1"><SelectValue /></SelectTrigger>
   521→                          <SelectContent>
   522→                            <SelectItem value="Hipertrofia">Hipertrofia</SelectItem>
