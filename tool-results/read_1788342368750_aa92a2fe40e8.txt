   149→  const [client, setClient] = useState<ClientData | null>(null)
   150→  const [loading, setLoading] = useState(true)
   151→  const [activeTab, setActiveTab] = useState(    view === 'sessions' || view === 'client-sessions' ? 'sessions' :    view === 'questionnaire' || view === 'client-questionnaire' ? 'questionnaire' :    view === 'training' || view === 'client-training' ? 'training' :    view === 'progress' || view === 'client-progress' ? 'progress' :    'profile'  )
   152→  const [editMode, setEditMode] = useState(false)
   153→  const [editForm, setEditForm] = useState<Record<string, string>>({})
   154→  const [trainerNote, setTrainerNote] = useState('')
   155→
   156→  // Training state
   157→  const [showTrainingForm, setShowTrainingForm] = useState(false)
   158→  const [trainingForm, setTrainingForm] = useState<Training>({ id: '', name: '', isActive: true, days: [] })
   159→  const [showSessionForm, setShowSessionForm] = useState(false)

   162→  const [qForm, setQForm] = useState<Record<string, number>>({})

   165→  const [measForm, setMeasForm] = useState<Record<string, string>>({})
   166→

   168→  const [editCredentials, setEditCredentials] = useState(false)
   169→  const [credForm, setCredForm] = useState({ email: '', password: '' })

   171→  const [credError, setCredError] = useState('')
   172→  const [credSuccess, setCredSuccess] = useState('')
   173→

   185→          goal: data.goal || '',
   186→          experienceLevel: data.experienceLevel || '',
   187→          injuries: data.injuries || '',
   188→          weeklyAvailability: data.weeklyAvailability || '',
   189→          notes: data.notes || '',
   190→        })
   191→      }
   192→    }).catch(() => {}).finally(() => {
   193→      if (!cancelled) setLoading(false)
   194→    })
   195→    return () => { cancelled = true }

   198→  const reloadClient = () => {
   199→    apiGet(`/api/clients/${clientId}`).then((data) => {
   200→      setClient(data)
   201→      if (data) {
   202→        setEditForm({
   203→          name: data.name || '',
   204→          age: data.age?.toString() || '',
   205→          height: data.height?.toString() || '',
   206→          weight: data.weight?.toString() || '',
   207→          goal: data.goal || '',
   208→          experienceLevel: data.experienceLevel || '',
   209→          injuries: data.injuries || '',
   210→          weeklyAvailability: data.weeklyAvailability || '',
   211→          notes: data.notes || '',
   212→        })
   213→      }
   214→    }).catch(() => {})
   215→  }
   216→

   218→    if (!client?.user?.id) return
   219→    setCredError('')
   220→    setCredSuccess('')
   221→    setSavingCreds(true)
   222→    try {
   223→      const payload: Record<string, string> = {}
   224→      if (credForm.email && credForm.email !== client.user.email) {
   225→        payload.email = credForm.email
   226→      }
   227→      if (credForm.password) {
   228→        payload.password = credForm.password
   229→      }
   230→      if (Object.keys(payload).length === 0) {
   231→        setCredError('No hay cambios que guardar')
   232→        setSavingCreds(false)
   233→        return
   234→      }
   235→      await apiPut(`/api/users/${client.user.id}`, payload)
   236→      setCredSuccess('Acceso actualizado correctamente')
   237→      setEditCredentials(false)
   238→      setCredForm({ email: '', password: '' })
   239→      reloadClient()

   241→      const msg = e instanceof Error ? e.message : 'Error al guardar'
   242→      if (msg.includes('409') || msg.includes('email')) {
   243→        setCredError('El email ya está en uso')
   244→      } else {
   245→        setCredError('Error al guardar los cambios')
   246→      }
   247→    } finally {
   248→      setSavingCreds(false)
   249→    }
   250→  }

   259→        goal: editForm.goal || null,
   260→        experienceLevel: editForm.experienceLevel || null,
   261→        injuries: editForm.injuries || null,
   262→        weeklyAvailability: editForm.weeklyAvailability || null,
   263→        notes: editForm.notes || null,
   264→      })
   265→      setEditMode(false)
   266→      reloadClient()
   267→    } catch {}
   268→  }
   269→
   270→  const addTrainerNote = async () => {
   271→    if (!trainerNote.trim()) return
   272→    try {
   273→      await apiPost(`/api/clients/${clientId}`, { action: 'addNote', content: trainerNote })
   274→      setTrainerNote('')
   275→      reloadClient()
   276→    } catch {}
   277→  }
   278→
   279→  // Training helpers

   283→      ...trainingForm,
   284→      days: [...trainingForm.days, { dayNumber: num, dayName: `Día ${num}`, exercises: [] }],
   285→    })

   289→    const days = [...trainingForm.days]
   290→    days[dayIdx] = {
   291→      ...days[dayIdx],
   292→      exercises: [...days[dayIdx].exercises, { exerciseName: '', order: days[dayIdx].exercises.length, prescribedSets: 3, prescribedReps: '8-12', prescribedRir: '1-2' }],
   293→    }

   302→    setTrainingForm({ ...trainingForm, days })
   303→  }
   304→

   306→    const days = [...trainingForm.days]
   307→    days[dayIdx] = { ...days[dayIdx], exercises: days[dayIdx].exercises.filter((_, i) => i !== exIdx) }
   308→    setTrainingForm({ ...trainingForm, days })
   309→  }
   310→
   311→  const removeTrainingDay = (dayIdx: number) => {
   312→    const days = trainingForm.days.filter((_, i) => i !== dayIdx).map((d, i) => ({ ...d, dayNumber: i + 1 }))
   313→    setTrainingForm({ ...trainingForm, days })
   314→  }

   317→    try {
   318→      const body = {
   319→        name: trainingForm.name,
   320→        mesocycle: trainingForm.mesocycle || 1,
   321→        week: trainingForm.week || 1,

   334→            prescribedTempo: e.prescribedTempo,
   335→            prescribedWeight: e.prescribedWeight,
   336→            notes: e.notes,
   337→          })),
   338→        })),
   339→      }
   340→      if (trainingForm.id) {

   342→      } else {
   343→        await apiPost('/api/trainings', body)
   344→      }
   345→      setShowTrainingForm(false)
   346→      setTrainingForm({ id: '', name: '', isActive: true, days: [] })
   347→      reloadClient()

   349→  }
   350→

   353→      await apiGet(`/api/trainings/${t.id}?duplicate=true`)
   354→      reloadClient()
   355→    } catch {}
   356→  }
   357→

   389→    const exercises = [...sessionForm.exercises]
   390→    const sets = [...exercises[exIdx].sets]
   391→    sets[setIdx] = { ...sets[setIdx], [field]: value }
   392→    exercises[exIdx] = { ...exercises[exIdx], sets }
   393→    setSessionForm({ ...sessionForm, exercises })

   396→  const saveSession = async () => {
   397→    try {
   398→      const activeTraining = client?.trainings.find(t => t.isActive)

   400→        clientId,

   405→          exerciseName: e.exerciseName,
   406→          order: e.order,
   407→          sets: e.sets.map((s) => ({
   408→            setNumber: s.setNumber,
   409→            reps: s.reps,
   410→            weight: s.weight,
   411→            rir: s.rir,
   412→            rpe: s.rpe,
   413→          })),
   414→        })),
   415→      })
   416→      setShowSessionForm(false)
   417→      setSessionForm({ id: '', date: new Date().toISOString(), exercises: [] })
   418→      reloadClient()

   426→        clientId,
   427→        sleepQuality: qForm.sleepQuality,
   428→        fatigue: qForm.fatigue,
   429→        stress: qForm.stress,
   430→        motivation: qForm.motivation,
   431→        pain: qForm.pain,
   432→        energy: qForm.energy,
   433→        hunger: qForm.hunger,
   434→        adherence: qForm.adherence,
   435→        trainingFeeling: qForm.trainingFeeling,
   436→        notes: qNotes,

   448→      await apiPost('/api/measurements', {
   449→        clientId,
   450→        weight: measForm.weight ? parseFloat(measForm.weight) : null,
   451→        chest: measForm.chest ? parseFloat(measForm.chest) : null,

   469→
   470→  const statusConfig = {
   471→    green: { label: 'Correcto', color: '#22c55e', bg: 'bg-green-50' },
   472→    yellow: { label: 'Revisar', color: '#eab308', bg: 'bg-yellow-50' },

   484→    if (!score) return 'bg-muted'
   485→    if (score >= 70) return 'bg-green-500'
   486→    if (score >= 40) return 'bg-yellow-500'
   487→    return 'bg-red-500'
   488→  }
   489→
   490→  if (loading) return <div className="flex items-center justify-center h-64 text-muted-foreground">Cargando...</div>

   493→  const sc = statusConfig[client.status as keyof typeof statusConfig] || statusConfig.red
   494→  const latestQ = client.questionnaires?.[0]
   495→  const weightData = client.measurements?.filter(m => m.weight).map(m => ({ date: new Date(m.date).toLocaleDateString('es-ES', { day: '2-digit', month: 'short' }), peso: m.weight })) || []
   496→  const volumeData = client.sessions?.slice(-10).map(s => ({ date: new Date(s.date).toLocaleDateString('es-ES', { day: '2-digit', month: 'short' }), volumen: s.totalVolume || 0 })) || []
   497→
   498→  return (
   499→    <div className="space-y-6">
   500→      {/* Header */}

   502→        <Button variant="ghost" size="icon" onClick={() => { selectClient(null); setView('clients') }}>
   503→          <ArrowLeft className="h-5 w-5" />
   504→        </Button>
   505→        <div className="flex-1 min-w-0">
   506→          <div className="flex items-center gap-3 flex-wrap">
   507→            <h1 className="text-2xl font-light tracking-wider text-[#334155]">{client.name}</h1>
   508→            <Badge variant="outline" className="text-xs" style={{ borderColor: sc.color, color: sc.color }}>{sc.label}</Badge>

   510→          <p className="text-sm text-muted-foreground mt-1">{client.goal || 'Sin objetivo'}{client.weight ? ` · ${client.weight} kg` : ''}</p>

   529→            key={tab.value}

   531→            className={`inline-flex items-center gap-1.5 rounded-md px-3 py-2 text-xs font-medium transition-colors ${activeTab === tab.value ? 'bg-[#0E7490] text-white shadow-sm' : 'text-muted-foreground hover:text-foreground hover:bg-muted'}`}

   538→
   539→        {/* PROFILE TAB */}
   540→      {activeTab === 'profile' && <div className="space-y-6 mt-6">
   541→          <div className="grid md:grid-cols-2 gap-6">
   542→            <Card className="border-0 shadow-sm">
